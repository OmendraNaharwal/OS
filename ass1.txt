
#!/bin/bash

F="addressbook.txt"

while true
do
    echo "1.Create 2.View 3.Insert 4.Delete 5.Modify 6.Exit"
    read -p "Enter choice: " c

    case $c in

    1)
        > $F
        echo "Created"
        ;;

    2)
        cat $F
        ;;

    3)
        read -p "Name: " n
        read -p "Phone: " p
        read -p "Email: " e
        echo "$n $p $e" >> $F
        ;;

    4)
        read -p "Name: " n
        grep -v "^$n " $F > temp
        mv temp $F
        echo "Deleted"
        ;;

    5)
        read -p "Name: " n
        read -p "Phone: " p
        read -p "Email: " e
        sed -i "s/^$n .*/$n $p $e/" $F
        echo "Updated"
        ;;

    6)
        exit
        ;;

    *)
        echo "Invalid Choice"
        ;;
    esac
done
